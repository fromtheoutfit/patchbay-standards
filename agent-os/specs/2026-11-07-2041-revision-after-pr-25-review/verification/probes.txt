"""Probe r: what Event.at's check, tzinfo.utcoffset(None) == timedelta(0), accepts and refuses."""

from datetime import UTC, datetime, timedelta, timezone, tzinfo
from zoneinfo import ZoneInfo

from patchbay.errors import PatchbayValidationError
from patchbay.events import Event


class ZeroForNone(tzinfo):
    """Answers zero for no particular instant, and an hour east of UTC at every real one."""

    def utcoffset(self, dt: datetime | None) -> timedelta:
        return timedelta(0) if dt is None else timedelta(hours=1)

    def dst(self, dt: datetime | None) -> None:
        return None

    def tzname(self, dt: datetime | None) -> str:
        return "ZERO-FOR-NONE"


parsed_z = Event.model_validate_json(
    '{"name": "thing.happened", "at": "2026-07-15T12:00:00Z", "source": "s", "correlation_id": "c"}'
).at
cases = [
    ("datetime.UTC", datetime(2026, 7, 15, 12, tzinfo=UTC)),
    ("timezone.utc", datetime(2026, 7, 15, 12, tzinfo=timezone.utc)),
    ("ZoneInfo('UTC')", datetime(2026, 7, 15, 12, tzinfo=ZoneInfo("UTC"))),
    ("pydantic's parsed Z", parsed_z),
    ("naive", datetime(2026, 7, 15, 12)),
    ("+01:00, fixed", datetime(2026, 7, 15, 12, tzinfo=timezone(timedelta(hours=1)))),
    ("Europe/London in January", datetime(2026, 1, 15, 12, tzinfo=ZoneInfo("Europe/London"))),
    ("Europe/London in July", datetime(2026, 7, 15, 12, tzinfo=ZoneInfo("Europe/London"))),
    ("ZeroForNone", datetime(2026, 7, 15, 12, tzinfo=ZeroForNone())),
]
for label, at in cases:
    zone = at.tzinfo
    asked = "naive" if zone is None else repr(zone.utcoffset(None))
    at_instant = "naive" if zone is None else repr(at.utcoffset())
    try:
        event = Event(name="thing.happened", at=at, source="s", correlation_id="c")
        print(f"r {label}: utcoffset(None) {asked}, at the instant {at_instant}: accepted, "
              f"held as {event.at.isoformat()}, tzinfo is UTC {event.at.tzinfo is UTC}, "
              f"same instant {event.at == at}")
    except PatchbayValidationError as error:
        print(f"r {label}: utcoffset(None) {asked}, at the instant {at_instant}: refused, {error}")

"""Probe s: int / int, which Python rounds correctly, and where it overflows."""

import sys

cases = (
    ("8 / 10**400", 8, 10**400),
    ("2**24 / 768000", 2**24, 768_000),
    ("2**1024 / 3", 2**1024, 3),
    ("2**1024 / 1", 2**1024, 1),
    ("10**400 / 1", 10**400, 1),
)
print(f"s the largest float is {sys.float_info.max!r}")
for label, numerator, denominator in cases:
    try:
        print(f"s {label}: {numerator / denominator!r}")
    except OverflowError as error:
        print(f"s {label}: OverflowError: {error}")
